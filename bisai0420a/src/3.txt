class DSUclzHpmdyj:
    def __init__(self, n):
        self.fathers = list(range(n))
        self.size = [1] * n  
        self.edge_size = [0] * n  
        self.n = n
        self.set_count = n  

    def find_fa(self, x):
        fs = self.fathers
        t = x
        while fs[x] != x:
            x = fs[x]
        while t != x:
            fs[t], t = x, fs[t]
        return x

    def union(self, x: int, y: int) -> bool:
        x = self.find_fa(x)
        y = self.find_fa(y)
        if x == y:
            self.edge_size[y] += 1
            return False
        self.fathers[x] = y
        self.size[y] += self.size[x]
        self.edge_size[y] += 1 + self.edge_size[x]
        self.set_count -= 1
        return True
